# libdbgshim for Android.
#
# Builds ONLY the src/shared and src/dbgshim subtrees of a dotnet/diagnostics
# checkout with the Android NDK toolchain. SOS, its lldb plugin and the test
# helpers are deliberately left out: dbgshim does not need them, and their
# build prerequisites (LLDB headers, the MSBuild "native prereqs" step whose
# Cdac.Transport package does not exist for the android RIDs) are what stop
# the upstream ./build.sh -os android from completing.
#
# This file mirrors the upstream top-level CMakeLists.txt of dotnet/diagnostics
# for the parts that matter to those two subtrees; keep it in step when the
# pinned diagnostics commit moves.
#
# Configure (arm64 example; the TARGET_BUILD_ARCH environment variable is read
# by eng/native/tryrun.cmake, which must be loaded with -C BEFORE the toolchain
# file and therefore cannot see ANDROID_ABI yet):
#
#   export TARGET_BUILD_ARCH=arm64
#   cmake -S <this dir> -B <build dir> \
#     -C <diagnostics>/eng/native/tryrun.cmake \
#     -DCMAKE_TOOLCHAIN_FILE=$NDK/build/cmake/android.toolchain.cmake \
#     -DANDROID_ABI=arm64-v8a -DANDROID_PLATFORM=android-33 \
#     -DCMAKE_BUILD_TYPE=Release \
#     -DDIAGNOSTICS_DIR=<diagnostics> \
#     -DHAVE_LTTNG_TRACEPOINT_H=1
#
# HAVE_LTTNG_TRACEPOINT_H=1 pre-answers the PAL's lttng probe. Nothing in the
# shared or dbgshim subtrees consumes that variable except the FATAL_ERROR
# that fires when it is false while FEATURE_EVENT_TRACE is on; and turning
# FEATURE_EVENT_TRACE off instead makes clrdefinitions.cmake register a dummy
# "eventing_headers" target once per include, which collides because both
# subtrees include shared.cmake. dotnet/runtime's own PAL keys the same probe
# to FEATURE_EVENTSOURCE_XPLAT, which is off for Android; this is the moral
# equivalent for the diagnostics copy of the PAL.

cmake_minimum_required(VERSION 3.15)

cmake_policy(SET CMP0042 NEW) # MACOSX_RPATH is enabled by default.
if(POLICY CMP0177)
  cmake_policy(SET CMP0177 NEW) # install() paths are normalized
endif(POLICY CMP0177)

project(dbgshim-android)

# Default to the vendored, pinned dbgshim source that sits next to this file
# (android/dbgshim/diagnostics-src). Override -DDIAGNOSTICS_DIR to use a full
# dotnet/diagnostics checkout instead.
set(DIAGNOSTICS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/diagnostics-src" CACHE PATH "Root of a dotnet/diagnostics checkout (or the vendored subset)")
if(NOT EXISTS "${DIAGNOSTICS_DIR}/src/dbgshim/CMakeLists.txt")
  message(FATAL_ERROR "DIAGNOSTICS_DIR must point at a dotnet/diagnostics checkout; '${DIAGNOSTICS_DIR}/src/dbgshim/CMakeLists.txt' was not found")
endif()

# include shared compiler configs (these compute CLR_* host/target variables
# from CMAKE_SYSTEM_NAME=Android and CMAKE_SYSTEM_PROCESSOR from the NDK)
include(${DIAGNOSTICS_DIR}/eng/native/configurepaths.cmake)
include(${CLR_ENG_NATIVE_DIR}/configurecompiler.cmake)

if(NOT CLR_CMAKE_TARGET_ANDROID)
  message(FATAL_ERROR "This wrapper targets Android only. Pass the NDK's android.toolchain.cmake as CMAKE_TOOLCHAIN_FILE.")
endif()

# The -fms-extensions enable the stuff like __if_exists, __declspec(uuid()), etc.
add_compile_options(-fms-extensions)
add_compile_options(-fvisibility=default)

# Register the default component
set(CMAKE_INSTALL_DEFAULT_COMPONENT_NAME diagnostics)
add_component(diagnostics)

set(ROOT_DIR ${DIAGNOSTICS_DIR})

# _version.c is normally generated by an MSBuild step (GenerateVersion) that
# cannot run for the android RIDs. Provide an equivalent stub carrying the
# checkout's commit hash so the embedded version string stays truthful.
if(NOT EXISTS "${VERSION_FILE_PATH}")
  execute_process(
    COMMAND git -C "${DIAGNOSTICS_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE DIAGNOSTICS_COMMIT
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET)
  if(NOT DIAGNOSTICS_COMMIT)
    set(DIAGNOSTICS_COMMIT "0000000000000000000000000000000000000000")
  endif()
  get_filename_component(VERSION_FILE_DIR "${VERSION_FILE_PATH}" DIRECTORY)
  file(MAKE_DIRECTORY "${VERSION_FILE_DIR}")
  file(WRITE "${VERSION_FILE_PATH}"
"// Generated by the CodeBrix dbgshim-android wrapper; the upstream generator is an MSBuild step that cannot run for android RIDs.
#include <stdint.h>
#ifdef __cplusplus
extern \"C\" {
#endif
static char sccsid[] = \"@(#)Version 0.0.0.0 @Commit: ${DIAGNOSTICS_COMMIT}\";
#ifdef __cplusplus
}
#endif
")
endif()

# Drop the static scope for SOS, so it's available outside the
# compilation unit for external linkage; see extern declaration
# in SOS sources. (Kept identical to upstream so the file stays usable
# by a full upstream build of the same checkout.)
file(READ "${VERSION_FILE_PATH}" VERSION_FILE_CONTENTS)
string(REPLACE "static char" "char" VERSION_LINE_WITHOUT_STATIC "${VERSION_FILE_CONTENTS}")
file(WRITE "${VERSION_FILE_PATH}" "${VERSION_LINE_WITHOUT_STATIC}")

set(CORECLR_SET_RPATH ON)
if(CORECLR_SET_RPATH)
  # Enable @rpath support for shared libraries.
  set(MACOSX_RPATH ON)
endif(CORECLR_SET_RPATH)

OPTION(CLR_CMAKE_ENABLE_CODE_COVERAGE "Enable code coverage" OFF)
OPTION(CLR_CMAKE_WARNINGS_ARE_ERRORS "Warnings are errors" ON)

#------------------------------------
# Definitions (for platform)
#-----------------------------------

if (CLR_CMAKE_HOST_ARCH_AMD64)
  add_definitions(-D_AMD64_)
  add_definitions(-D_WIN64)
  add_definitions(-DAMD64)
  add_definitions(-DBIT64=1)
elseif (CLR_CMAKE_HOST_ARCH_ARM64)
  add_definitions(-D_ARM64_)
  add_definitions(-DARM64)
  add_definitions(-D_WIN64)
  add_definitions(-DBIT64=1)
else ()
  clr_unknown_arch()
endif ()

if (CLR_CMAKE_TARGET_ARCH_AMD64)
  add_definitions(-DDBG_TARGET_AMD64_UNIX)
  add_definitions(-D_TARGET_64BIT_=1)
  add_definitions(-D_TARGET_AMD64_=1)
  add_definitions(-DDBG_TARGET_64BIT=1)
  add_definitions(-DDBG_TARGET_AMD64=1)
  add_definitions(-DDBG_TARGET_WIN64=1)
elseif (CLR_CMAKE_TARGET_ARCH_ARM64)
  add_definitions(-DDBG_TARGET_ARM64_UNIX)
  add_definitions(-D_TARGET_ARM64_=1)
  add_definitions(-D_TARGET_64BIT_=1)
  add_definitions(-DDBG_TARGET_64BIT=1)
  add_definitions(-DDBG_TARGET_ARM64=1)
  add_definitions(-DDBG_TARGET_WIN64=1)
  add_definitions(-DFEATURE_MULTIREG_RETURN)
else ()
  clr_unknown_arch()
endif (CLR_CMAKE_TARGET_ARCH_AMD64)

#--------------------------------------
# FEATURE Defines
#--------------------------------------

add_definitions(-DFEATURE_PAL)

#-----------------------------------------
# Native Projects (the subset dbgshim needs)
#-----------------------------------------
include_directories(${DIAGNOSTICS_DIR}/src/inc)
include_directories(${DIAGNOSTICS_DIR}/src/inc/rt)

add_subdirectory(${DIAGNOSTICS_DIR}/src/shared shared)
add_subdirectory(${DIAGNOSTICS_DIR}/src/dbgshim dbgshim)
